// ═══════════════════════════════════════════════ // Alma Plus TV — Monitor Editorial // Apps Script — pega este código en script.google.com // ═══════════════════════════════════════════════ const SHEET_ID = ''; // ← Pega aquí el ID de tu Google Sheet (opcional) const DATA_KEY = 'alma_data'; // clave en PropertiesService function doGet(e) { const action = e && e.parameter && e.parameter.action; if (action === 'pull') { return pullData(); } return ContentService .createTextOutput(JSON.stringify({status:'ok',message:'Alma Monitor API'})) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); if (payload.action === 'push') { return pushData(payload); } return jsonResponse({status:'error', message:'Acción desconocida'}); } catch(err) { return jsonResponse({status:'error', message: err.message}); } } function pushData(payload) { const store = PropertiesService.getScriptProperties(); // Guardamos en chunks si supera el límite de 9KB por propiedad const json = JSON.stringify({ notes: payload.notes || [], cobs: payload.cobs || [], criterios: payload.criterios || [], reds: payload.reds || [], timestamp: payload.timestamp || new Date().toISOString() }); // Dividir en chunks de 8000 chars const chunkSize = 8000; const chunks = Math.ceil(json.length / chunkSize); store.setProperty(DATA_KEY + '_chunks', String(chunks)); for (let i = 0; i < chunks; i++) { store.setProperty(DATA_KEY + '_' + i, json.slice(i * chunkSize, (i+1) * chunkSize)); } // También escribir en Sheet si está configurado if (SHEET_ID) writeToSheet(payload.notes || []); return jsonResponse({status:'ok', saved: (payload.notes||[]).length + ' notas'}); } function pullData() { const store = PropertiesService.getScriptProperties(); const chunks = parseInt(store.getProperty(DATA_KEY + '_chunks') || '0'); if (chunks === 0) { return jsonResponse({status:'ok', data:{notes:[],cobs:[],criterios:[],reds:[]}}); } let json = ''; for (let i = 0; i < chunks; i++) { json += store.getProperty(DATA_KEY + '_' + i) || ''; } const data = JSON.parse(json); return jsonResponse({status:'ok', data}); } function writeToSheet(notes) { try { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName('Contenidos'); if (!sheet) sheet = ss.insertSheet('Contenidos'); sheet.clearContents(); const headers = [ 'id','tipo','fecha','hora','postId','titulo','editor','autor', 'cat','reg','cob','vistas','gaPos','gaVis','gaInt', 'ventana','recursos','parrafos','calidad','elementos','rrss','obs' ]; const rows = notes.map(n => headers.map(h => { const v = n[h]; return Array.isArray(v) ? v.join('|') : (v == null ? '' : v); })); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); } } catch(err) { Logger.log('writeToSheet error: ' + err.message); } } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
